# 任務名稱
畢業預警報表排序調整

# 目標
讓 btnReport_Click 產生報表時，SelectedReportStudentList 的順序
與 dgDataGW 畫面目前顯示順序完全一致（包含使用者排序結果）。

# 問題原因
1. 使用 dgDataGW.SelectedRows，順序不等於畫面順序
2. 使用 OrderBy(ClassName).ThenBy(SeatNo) 重新排序，覆蓋 UI 排序

# 修改策略
1. 改用 dgDataGW.Rows（依畫面順序）
2. 判斷 row.Selected 取代 SelectedRows
3. 移除 OrderBy 排序

# 修改檔案
frmMain.cs

# 修改位置
btnReport_Click()

# 修改內容

## 1. 移除排序（務必刪除）
刪除以下程式碼：

SelectedReportStudentList = SelectedReportStudentList
    .OrderBy(x => x.ClassName)
    .ThenBy(x =>
    {
        int seatNo;
        return int.TryParse(x.SeatNo, out seatNo) ? seatNo : 0;
    }).ToList();

## 2. 修改資料來源邏輯（完整取代）

SelectedReportStudentList.Clear();

// 無選取 → 全部依畫面順序
if (dgDataGW.SelectedRows.Count == 0)
{
    foreach (DataGridViewRow row in dgDataGW.Rows)
    {
        if (row.IsNewRow)
            continue;

        ReportStudentInfo rs = row.Tag as ReportStudentInfo;
        if (rs != null)
            SelectedReportStudentList.Add(rs);
    }
}
else
{
    // 有選取 → 依畫面順序 + 判斷 Selected
    foreach (DataGridViewRow row in dgDataGW.Rows)
    {
        if (row.IsNewRow)
            continue;

        if (!row.Selected)
            continue;

        ReportStudentInfo rs = row.Tag as ReportStudentInfo;
        if (rs != null)
            SelectedReportStudentList.Add(rs);
    }
}

# 執行報表（保留原邏輯）
if (SelectedReportStudentList.Count > 0)
{
    bgwGrandCheckReport.RunWorkerAsync();
}
else
{
    MsgBox.Show("沒有資料。");
    return;
}

# 注意事項
1. 不可再加入任何 OrderBy（會破壞 UI 排序）
2. DataGridView.Rows 順序 = 畫面排序結果
3. SelectedRows 順序不可靠，禁止使用

# 驗證方式
1. 點擊 dgDataGW 欄位排序（姓名 / 班級 / 座號）
2. 不選資料 → 報表順序需一致
3. 選部分資料 → 報表順序需一致
4. 測試座號亂序（1,10,2）→ 不應被重排
